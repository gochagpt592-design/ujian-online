Integrasi Spreadsheet

Pengaturan & Kirim Key / Nilai ke Google Sheets

Panduan Penting Agar Tidak Keluar Halaman "Library Documentation":
  1. Buka Google Sheets Baru (sheets.new) lalu klik menu Ekstensi (Extensions) → Apps Script (Catatan: Kode baru di bawah juga sudah mendukung pembuatan Google Sheet otomatis walaupun Anda membukanya dari script.google.com).
  2. Klik "📋 Lihat & Salin Kode Apps Script (Code.gs)" di bawah, hapus kode lama di Code.gs, lalu tempelkan kode baru.
  3. Klik tombol Terapkan (Deploy) → Deployment baru (New deployment) → klik ikon roda gigi ⚙️ di kiri atas → WAJIB pilih "Aplikasi web" (Web app), JANGAN pilih "Library (Pustaka)"!
  4. Pada bagian Siapa yang memiliki akses (Who has access) pilih Siapa saja (Anyone) → klik Terapkan (Deploy) → salin URL Aplikasi Web yang berakhiran /exec (bukan /macros/library/...).
✓ Status: URL Google Apps Script sudah tertanam permanen di kode web. Otomatis aktif di semua browser/komputer sekolah!

// ============================================================================
// GOOGLE APPS SCRIPT (Code.gs) — OTOMATIS BUAT & BUKA GOOGLE SHEETS PER MAPEL
// Bekerja baik dari dalam Google Sheets maupun langsung dari script.google.com!
// PENTING SAAT DEPLOY: Pilih jenis "Aplikasi Web (Web App)", JANGAN pilih "Library"
// ============================================================================

function ambilAtauBuatSpreadsheet() {
  var ss = null;
  try {
    ss = SpreadsheetApp.getActiveSpreadsheet();
  } catch (e) {}
  if (ss) return ss;

  var props = PropertiesService.getScriptProperties();
  var savedId = props.getProperty('CBT_SPREADSHEET_ID');
  if (savedId) {
    try {
      ss = SpreadsheetApp.openById(savedId);
      if (ss) return ss;
    } catch (e) {}
  }

  // Jika dibuat dari script.google.com (bukan dari dalam Google Sheet), otomatis buatkan Google Sheet baru!
  ss = SpreadsheetApp.create('Database_Ujian_CBT_Terpisah_Per_Mapel');
  props.setProperty('CBT_SPREADSHEET_ID', ss.getId());
  return ss;
}

  return ss.getUrl();
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(15000);
  try {
    var raw = (e && e.postData && e.postData.contents) ? e.postData.contents : '{}';
    var data = {};
    try {
      data = JSON.parse(raw || '{}');
    } catch (parseErr) {
      data = {};
    }

    var ss = ambilAtauBuatSpreadsheet();
    var sentAt = data.sentAt || new Date().toISOString();

    // Kelompokkan data per Mata Pelajaran / Tugas agar Tabel Tugas 1 & Tugas 2 TIDAK TERCAMPUR
    var subjects = Array.isArray(data.subjects) ? data.subjects : [];
    if (subjects.length === 0 && Array.isArray(data.allData) && data.allData.length > 0) {
      var groupedMap = {};
      var groupedOrder = [];
      data.allData.forEach(function(r) {
        var mp = r.mataPelajaran || 'Mata Pelajaran 1';
        if (!groupedMap[mp]) {
          groupedMap[mp] = {
            orderNo: groupedOrder.length + 1,
            courseName: mp,
            supervisorPassword: r.keyPengawas || '-',
            rows: []
          };
          groupedOrder.push(groupedMap[mp]);
        }
        groupedMap[mp].rows.push(r);
      });
      subjects = groupedOrder;
    }

    // 1. Sheet Utama: Data_CBT_Lengkap (Tabel Terpisah Rapi per Tugas / Mata Pelajaran)
    var mainSheet = siapkanSheetBersih(ss, 'Data_CBT_Lengkap');
    var mainHeaders = [
      'No', 'NIM', 'Nama Mahasiswa', 'Program Studi', 'Kelas',
      'Tugas / Mata Pelajaran', 'Key Unik Mahasiswa (Menu 3)', 'Sandi Pengawas',
      'Status Ujian', 'Betul', 'Salah', 'Kosong', 'Nilai Akhir', 'Waktu Pembaruan'
    ];
    var mainWidths = [48, 115, 190, 165, 85, 210, 175, 130, 140, 68, 68, 68, 95, 165];
    renderBlokTabelPerMapel(mainSheet, subjects, mainHeaders, mainWidths, [2, 7, 8], [1, 5, 9, 10, 11, 12, 13], function(r, idx, subTitle) {
      return [
        idx + 1,
        String(r.nim || ''),
        String(r.nama || ''),
        String(r.prodi || ''),
        String(r.kelas || ''),
        String(subTitle || r.mataPelajaran || ''),
        String(r.keyUjian || ''),
        String(r.keyPengawas || ''),
        String(r.status || 'Belum Mengerjakan'),
        Number(r.betul || 0),
        Number(r.salah || 0),
        Number(r.kosong || 0),
        (r.nilai !== undefined && r.nilai !== '' && r.nilai !== null) ? Number(r.nilai) : '-',
        String(r.waktuSelesai || sentAt)
      ];
    });

    // 2. Sheet Daftar_Key_Ujian (Dipisah per Tugas / Mata Pelajaran)
    var keySheet = siapkanSheetBersih(ss, 'Daftar_Key_Ujian');
    var keyHeaders = ['No', 'NIM', 'Nama Mahasiswa', 'Kelas', 'Tugas / Mata Pelajaran', 'Key Unik Mahasiswa', 'Sandi Pengawas', 'Waktu Update'];
    var keyWidths = [48, 120, 200, 90, 220, 175, 135, 165];
    renderBlokTabelPerMapel(keySheet, subjects, keyHeaders, keyWidths, [2, 6, 7], [1, 4], function(r, idx, subTitle) {
      return [
        idx + 1,
        String(r.nim || ''),
        String(r.nama || ''),
        String(r.kelas || ''),
        String(subTitle || r.mataPelajaran || ''),
        String(r.keyUjian || ''),
        String(r.keyPengawas || ''),
        String(sentAt)
      ];
    });

    // 3. Sheet Hasil_Nilai_Ujian (Dipisah per Tugas / Mata Pelajaran)
    var scoreSheet = siapkanSheetBersih(ss, 'Hasil_Nilai_Ujian');
    var scoreHeaders = ['No', 'NIM', 'Nama Mahasiswa', 'Program Studi', 'Kelas', 'Tugas / Mata Pelajaran', 'Status Ujian', 'Betul', 'Salah', 'Kosong', 'Nilai Akhir', 'Waktu Selesai'];
    var scoreWidths = [48, 120, 195, 165, 85, 215, 140, 70, 70, 70, 95, 165];
    renderBlokTabelPerMapel(scoreSheet, subjects, scoreHeaders, scoreWidths, [2], [1, 5, 7, 8, 9, 10, 11], function(r, idx, subTitle) {
      return [
        idx + 1,
        String(r.nim || ''),
        String(r.nama || ''),
        String(r.prodi || ''),
        String(r.kelas || ''),
        String(subTitle || r.mataPelajaran || ''),
        String(r.status || 'Belum Mengerjakan'),
        Number(r.betul || 0),
        Number(r.salah || 0),
        Number(r.kosong || 0),
        (r.nilai !== undefined && r.nilai !== '' && r.nilai !== null) ? Number(r.nilai) : '-',
        String(r.waktuSelesai || '-')
      ];
    });

    // 4. Buat Sheet Tab Khusus untuk Masing-Masing Tugas / Mata Pelajaran (Tugas 1, Tugas 2, dst.)
    var activeTabNames = {};
    subjects.forEach(function(sub, sIdx) {
      var cleanName = String(sub.courseName || ('Mapel ' + (sIdx + 1))).replace(/[\[\]\*\/\?\:\\]/g, ' ').trim();
      var tabName = ('Tugas ' + (sIdx + 1) + ' - ' + cleanName).substring(0, 95);
      activeTabNames[tabName] = true;
      var subSheet = siapkanSheetBersih(ss, tabName);
      renderBlokTabelPerMapel(subSheet, [sub], mainHeaders, mainWidths, [2, 7, 8], [1, 5, 9, 10, 11, 12, 13], function(r, idx, subTitle) {
        return [
          idx + 1,
          String(r.nim || ''),
          String(r.nama || ''),
          String(r.prodi || ''),
          String(r.kelas || ''),
          String(subTitle || r.mataPelajaran || ''),
          String(r.keyUjian || ''),
          String(r.keyPengawas || ''),
          String(r.status || 'Belum Mengerjakan'),
          Number(r.betul || 0),
          Number(r.salah || 0),
          Number(r.kosong || 0),
          (r.nilai !== undefined && r.nilai !== '' && r.nilai !== null) ? Number(r.nilai) : '-',
          String(r.waktuSelesai || sentAt)
        ];
      });
    });

    // Bersihkan tab Tugas lama yang sudah dihapus di aplikasi
    var allSheets = ss.getSheets();
    allSheets.forEach(function(sh) {
      var nm = sh.getName();
      if (/^Tugas \d+ - /.test(nm) && !activeTabNames[nm] && ss.getSheets().length > 1) {
        ss.deleteSheet(sh);
      }
    });

    SpreadsheetApp.flush();
    return ContentService.createTextOutput(
      JSON.stringify({ status: 'ok', subjectsCount: subjects.length })
    ).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(
      JSON.stringify({ status: 'error', message: String(err) })
    ).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function siapkanSheetBersih(ss, name) {
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
  }
  try {
    if (sheet.getFilter()) sheet.getFilter().remove();
  } catch (e) {}
  try {
    var bandings = sheet.getBandings();
    for (var i = 0; i < bandings.length; i++) bandings[i].remove();
  } catch (e) {}
  sheet.clear();
  return sheet;
}

function renderBlokTabelPerMapel(sheet, subjects, headers, colWidths, textCols, centerCols, rowBuilder) {
  var numCols = headers.length;
  var currentRow = 1;

  if (!subjects || subjects.length === 0) {
    var emptyBanner = new Array(numCols).fill('');
    emptyBanner[0] = 'BELUM ADA DATA MATA PELAJARAN / UJIAN AKTIF';
    sheet.getRange(1, 1, 1, numCols).setValues([emptyBanner]);
    sheet.getRange(1, 1, 1, numCols).merge().setBackground('#111827').setFontColor('#ffffff').setFontWeight('bold').setHorizontalAlignment('left');
    sheet.getRange(2, 1, 1, numCols).setValues([headers]).setBackground('#e5e7eb').setFontColor('#000000').setFontWeight('bold').setHorizontalAlignment('center').setBorder(true, true, true, true, true, true, '#000000', SpreadsheetApp.BorderStyle.SOLID);
    sheet.setRowHeight(1, 28);
    sheet.setRowHeight(2, 26);
    for (var c = 0; c < colWidths.length; c++) sheet.setColumnWidth(c + 1, colWidths[c]);
    return;
  }

  for (var sIdx = 0; sIdx < subjects.length; sIdx++) {
    var sub = subjects[sIdx];
    var orderLabel = sub.orderNo || (sIdx + 1);
    var subTitle = 'TUGAS / UJIAN ' + orderLabel + ': ' + (sub.courseName || ('Mata Pelajaran ' + orderLabel));
    var rows = Array.isArray(sub.rows) ? sub.rows : [];

    // Baris Judul Tabel per Mata Pelajaran
    var bannerRow = new Array(numCols).fill('');
    bannerRow[0] = subTitle + '   |   Sandi Pengawas: ' + (sub.supervisorPassword || '-') + '   |   Peserta: ' + rows.length + ' Mahasiswa';
    var bannerRange = sheet.getRange(currentRow, 1, 1, numCols);
    bannerRange.setValues([bannerRow]);
    bannerRange.merge()
      .setBackground('#111827')
      .setFontColor('#ffffff')
      .setFontWeight('bold')
      .setFontSize(10)
      .setVerticalAlignment('middle')
      .setHorizontalAlignment('left')
      .setBorder(true, true, true, true, false, false, '#000000', SpreadsheetApp.BorderStyle.SOLID);
    sheet.setRowHeight(currentRow, 30);
    currentRow++;

    // Baris Header Kolom
    var headerRange = sheet.getRange(currentRow, 1, 1, numCols);
    headerRange.setValues([headers])
      .setBackground('#e5e7eb')
      .setFontColor('#000000')
      .setFontWeight('bold')
      .setFontSize(10)
      .setVerticalAlignment('middle')
      .setHorizontalAlignment('center')
      .setWrap(true)
      .setBorder(true, true, true, true, true, true, '#000000', SpreadsheetApp.BorderStyle.SOLID);
    sheet.setRowHeight(currentRow, 28);
    currentRow++;

    // Baris Isi Data Mahasiswa untuk Mata Pelajaran Ini
    if (rows.length > 0) {
      var matrix = [];
      for (var rIdx = 0; rIdx < rows.length; rIdx++) {
        matrix.push(rowBuilder(rows[rIdx], rIdx, subTitle));
      }
      var dataRange = sheet.getRange(currentRow, 1, matrix.length, numCols);
      // Format kolom teks (@) terlebih dahulu agar NIM/Key tidak berubah jadi angka eksponen
      for (var t = 0; t < textCols.length; t++) {
        sheet.getRange(currentRow, textCols[t], matrix.length, 1).setNumberFormat('@');
      }
      dataRange.setValues(matrix)
        .setFontSize(10)
        .setFontColor('#000000')
        .setVerticalAlignment('middle')
        .setBorder(true, true, true, true, true, true, '#9ca3af', SpreadsheetApp.BorderStyle.SOLID);

      // Warna selang-seling (zebra) agar tabel mudah dibaca
      var bgColors = [];
      for (var z = 0; z < matrix.length; z++) {
        var rowBg = (z % 2 === 0) ? '#ffffff' : '#f9fafb';
        bgColors.push(new Array(numCols).fill(rowBg));
        sheet.setRowHeight(currentRow + z, 24);
      }
      dataRange.setBackgrounds(bgColors);

      // Rata tengah untuk kolom angka & status
      for (var k = 0; k < centerCols.length; k++) {
        sheet.getRange(currentRow, centerCols[k], matrix.length, 1).setHorizontalAlignment('center');
      }
      currentRow += matrix.length;
    } else {
      var emptyRow = new Array(numCols).fill('');
      emptyRow[0] = 'Belum ada mahasiswa terdaftar pada mata pelajaran ini.';
      var emptyRange = sheet.getRange(currentRow, 1, 1, numCols);
      emptyRange.setValues([emptyRow]);
      emptyRange.merge().setHorizontalAlignment('center').setFontColor('#6b7280').setBorder(true, true, true, true, false, false, '#9ca3af', SpreadsheetApp.BorderStyle.SOLID);
      sheet.setRowHeight(currentRow, 24);
      currentRow++;
    }

    // Beri 1 baris jeda kosong antar tabel Mata Pelajaran agar tidak tercampur
    currentRow += 1;
  }

  for (var w = 0; w < colWidths.length; w++) {
    sheet.setColumnWidth(w + 1, colWidths[w]);
  }
}
